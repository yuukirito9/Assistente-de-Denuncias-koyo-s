// ==UserScript==
// @name         Assistente de Denúncias Koyo's
// @namespace    grepolis-support-helper
// @version      2.11.2
// @description  Adiciona captura de denúncias pelo Grepolis e GrepoData, mantendo o fluxo normal do suporte.
// @match        https://support.innogames.com/playerticket/new*
// @match        https://support.innogames.com/t*
// @match        https://*.grepolis.com/game*
// @match        https://grepodata.com/player*
// @match        https://grepodata.com/alliance*
// @grant        GM_getValue
// @grant        GM_setValue
// @run-at       document-idle
// ==/UserScript==

(() => {
    'use strict';

    const CHAVE_JOGADORES = 'gp_multicontas_registros_v2';
    const CHAVE_JOGADORES_ANTIGA = 'gp_multicontas_jogadores_v1';
    const CHAVE_SELECAO = 'gp_multicontas_selecao_v1';
    const CHAVE_MODO_SELECAO = 'gp_multicontas_modo_selecao_v1';
    const CHAVE_ACAO_PENDENTE = 'gp_multicontas_acao_pendente_v1';
    const CHAVE_ULTIMA_CAPTURA = 'gp_multicontas_ultima_captura_v1';

    // Organização dos jogadores: grupo de origem + estado do relatório.
    // O grupo identifica a origem/contexto; o estado identifica se o jogador
    // já passou por um lote confirmado no suporte.
    const GRUPO_LEGADO = {
        tipo: 'legado',
        nome: 'Sem aliança / denúncia única'
    };
    const GRUPO_PERFIL = {
        tipo: 'perfil',
        nome: 'Adicionados por perfil'
    };

    // Armazenamento compartilhado entre o jogo e a página de suporte.
    function normalizarTexto(nome) {
        return String(nome || '').replace(/\s+/g, ' ').trim();
    }

    function criarGrupo(tipo = 'legado', nome = GRUPO_LEGADO.nome) {
        return {
            tipo: tipo || 'legado',
            nome: normalizarTexto(nome) || GRUPO_LEGADO.nome
        };
    }

    function normalizarRegistro(item) {
        if (typeof item === 'string') {
            const nome = normalizarTexto(item);
            return nome
                ? {
                    nome,
                    tipo: 'jogador',
                    grupo: { ...GRUPO_LEGADO },
                    estado: 'pendente',
                    reportes: 0
                }
                : null;
        }

        if (!item || typeof item !== 'object') return null;

        const nome = normalizarTexto(item.nome);
        if (!nome) return null;

        const grupo = item.grupo && typeof item.grupo === 'object'
            ? criarGrupo(item.grupo.tipo, item.grupo.nome)
            : item.pasta && typeof item.pasta === 'object'
                ? criarGrupo(item.pasta.tipo, item.pasta.nome)
                : item.origem && typeof item.origem === 'object'
                    ? criarGrupo(item.origem.tipo, item.origem.nome)
                    : { ...GRUPO_LEGADO };

        const estado = item.estado === 'reportado'
            ? 'reportado'
            : 'pendente';

        return {
            nome,
            tipo: item.tipo === 'alianca' ? 'alianca' : 'jogador',
            grupo,
            estado,
            reportes: Number.isFinite(Number(item.reportes))
                ? Math.max(0, Math.floor(Number(item.reportes)))
                : 0
        };
    }

    function lerJogadoresGuardados() {
        try {
            let valor = GM_getValue(CHAVE_JOGADORES, null);
            const migrando = !Array.isArray(valor);
            if (migrando) valor = GM_getValue(CHAVE_JOGADORES_ANTIGA, []);
            if (!Array.isArray(valor)) return [];

            const registros = valor.map(normalizarRegistro).filter(Boolean);

            // Também normaliza registros antigos que já estavam na chave v2.
            const mudouFormato = registros.some((registro, indice) => {
                const original = valor[indice];
                return typeof original === 'string' ||
                    !original ||
                    original.grupo?.nome !== registro.grupo.nome ||
                    original.grupo?.tipo !== registro.grupo.tipo ||
                    (original.estado !== 'reportado' && original.estado !== 'pendente') ||
                    !Number.isFinite(Number(original.reportes)) || Number(original.reportes) < 0;
            });

            if (migrando || mudouFormato) {
                GM_setValue(CHAVE_JOGADORES, registros);
            }

            return registros;
        } catch (erro) {
            console.error('[Grepolis] Erro ao ler a lista de jogadores:', erro);
            return [];
        }
    }

    function salvarJogadoresGuardados(registros) {
        GM_setValue(CHAVE_JOGADORES, registros);
    }

    // Ao abrir um ticket existente (/t + número), sincroniza os jogadores
    // exibidos pelo próprio suporte com a lista local do assistente.
    function sincronizarJogadoresDoTicket() {
        if (!/^\/t\d+\/?$/.test(window.location.pathname)) return false;

        const paragrafo = [...document.querySelectorAll('p.text-primary')]
            .find(elemento => /Jogadores denunciados\s*:/i.test(elemento.textContent || ''));
        if (!paragrafo) return true;

        const texto = paragrafo.textContent || '';
        const correspondencia = texto.match(/Jogadores denunciados\s*:\s*(.*)/i);
        if (!correspondencia) return true;

        const nomes = [...new Set(correspondencia[1]
            .split(/[,;\n]+/)
            .map(nome => normalizarTexto(nome))
            .filter(Boolean))];
        if (!nomes.length) return true;

        const registros = lerJogadoresGuardados();
        const porNome = new Map(registros.map(registro => [
            registro.nome.toLocaleLowerCase(), registro
        ]));

        for (const nome of nomes) {
            const chave = nome.toLocaleLowerCase();
            const existente = porNome.get(chave);
            if (existente) {
                existente.estado = 'reportado';
            } else {
                const novo = {
                    nome,
                    tipo: 'jogador',
                    grupo: { ...GRUPO_LEGADO },
                    estado: 'reportado'
                };
                registros.push(novo);
                porNome.set(chave, novo);
            }
        }

        salvarJogadoresGuardados(registros);
        console.info(`[Grepolis] Ticket ${window.location.pathname} sincronizado: ${nomes.length} jogador(es) marcado(s) como reportado(s).`);
        return true;
    }

    // A página de ticket não precisa carregar o painel do assistente; basta
    // sincronizar os nomes e encerrar a execução nesta rota.
    if (sincronizarJogadoresDoTicket()) return;

    function chaveJogador(nome) {
        return normalizarNome(nome).toLocaleLowerCase();
    }

    function adicionarNomes(nomes, grupo = GRUPO_LEGADO) {
        const registros = lerJogadoresGuardados();
        const porNome = new Map(
            registros.map(registro => [chaveJogador(registro.nome), registro])
        );
        const novosRegistros = [];
        let adicionados = 0;

        const grupoAtual = criarGrupo(grupo.tipo, grupo.nome);

        for (const bruto of nomes) {
            const nome = normalizarTexto(bruto);
            if (!nome) continue;

            const chave = chaveJogador(nome);
            const existente = porNome.get(chave);

            if (existente) {
                // Capturar pela aliança associa o jogador à pasta dessa aliança,
                // mesmo que ele já estivesse cadastrado por outra fonte. A
                // identidade continua única e não criamos uma segunda entrada.
                // Capturas individuais só retiram o jogador do grupo legado.
                if (
                    (grupoAtual.tipo === 'alianca' &&
                        (existente.grupo?.tipo !== grupoAtual.tipo ||
                         chaveJogador(existente.grupo?.nome) !== chaveJogador(grupoAtual.nome))) ||
                    (existente.grupo?.tipo === GRUPO_LEGADO.tipo &&
                        grupoAtual.tipo !== GRUPO_LEGADO.tipo)
                ) {
                    existente.grupo = { ...grupoAtual };
                }
                continue;
            }

            const registro = {
                nome,
                tipo: 'jogador',
                grupo: { ...grupoAtual },
                estado: 'pendente'
            };

            porNome.set(chave, registro);
            novosRegistros.push(registro);
            adicionados++;
        }

        const capturaAtual = [...new Map(
            nomes.map(nome => [chaveJogador(nome), normalizarTexto(nome)])
        ).values()].filter(Boolean);

        GM_setValue(CHAVE_ULTIMA_CAPTURA, capturaAtual);
        salvarJogadoresGuardados([...novosRegistros, ...registros]);

        return {
            total: registros.length + adicionados,
            adicionados,
            registros: lerJogadoresGuardados()
        };
    }

    // Incrementa apenas quando o fluxo do formulário confirma uma nova inclusão.
    // Abrir tickets antigos ou apenas marcar o estado como reportado não altera a contagem.
    function incrementarContagemReportes(nome) {
        const chave = chaveJogador(nome);
        const registros = lerJogadoresGuardados();
        const registro = registros.find(item => chaveJogador(item.nome) === chave);

        if (!registro) {
            console.warn(`[Grepolis] Não foi possível contar o reporte: jogador não cadastrado (${nome}).`);
            return false;
        }

        registro.reportes = Math.max(0, Math.floor(Number(registro.reportes) || 0)) + 1;
        salvarJogadoresGuardados(registros);
        return true;
    }

    function marcarJogadoresComoReportados(nomes) {
        const chaves = new Set(nomes.map(chaveJogador));
        const registros = lerJogadoresGuardados();

        let alterados = 0;

        for (const registro of registros) {
            if (chaves.has(chaveJogador(registro.nome)) &&
                registro.estado !== 'reportado') {
                registro.estado = 'reportado';
                alterados++;
            }
        }

        if (alterados) salvarJogadoresGuardados(registros);
        return alterados;
    }

    function iniciarColetorNoJogo() {
        // O assistente não fica mais fixo na tela: o botão é criado
        // somente dentro de um perfil individual ou de aliança aberto.
        const ID_BOTAO = 'gp-denunciar-perfil-koyos';

        function encontrarPerfilJogador() {
            const cabecalhos = [...document.querySelectorAll('#player_info.bold h3')]
                .filter(el => el.getClientRects().length && getComputedStyle(el).display !== 'none');

            for (const cabecalho of cabecalhos.reverse()) {
                const info = cabecalho.closest('#player_info.bold');
                if (!info) continue;
                const raiz = info.closest('.ui-dialog, .gpwindow') || info;
                if (raiz.querySelector('.members_list')) continue;
                const nome = normalizarTexto(cabecalho.textContent);
                if (nome) return { tipo: 'jogador', nome, raiz: info };
            }
            return null;
        }

        function encontrarPerfilAlianca() {
            const cabecalhos = [...document.querySelectorAll('#player_info h3.bold')]
                .filter(el => el.getClientRects().length && getComputedStyle(el).display !== 'none');

            for (const cabecalho of cabecalhos.reverse()) {
                const info = cabecalho.closest('#player_info');
                if (!info || info.classList.contains('bold')) continue;
                const nome = normalizarTexto(cabecalho.textContent);
                if (nome) return { tipo: 'alianca', nome, raiz: info };
            }
            return null;
        }

        function membrosDaAlianca(raiz) {
            // Tenta primeiro a lista dentro do perfil aberto e depois a janela
            // visível que contém a lista de membros da aliança.
            const janela = raiz.closest('.ui-dialog, .gpwindow') || raiz;
            let links = [...janela.querySelectorAll('.members_list a.gp_player_link')];

            if (!links.length) {
                const janelas = [...document.querySelectorAll('.ui-dialog, .gpwindow')]
                    .filter(el => el.getClientRects().length && getComputedStyle(el).display !== 'none');
                for (const outra of janelas) {
                    links = [...outra.querySelectorAll('.members_list a.gp_player_link')];
                    if (links.length) break;
                }
            }

            const nomes = links
                .map(link => normalizarTexto(link.getAttribute('title') || link.textContent))
                .filter(Boolean);
            return [...new Map(nomes.map(nome => [nome.toLocaleLowerCase(), nome])).values()];
        }

        function abrirSuporteComCaptura(perfil) {
            let nomes = [];
            if (perfil.tipo === 'jogador') {
                nomes = [perfil.nome];
            } else {
                nomes = membrosDaAlianca(perfil.raiz);
                // Se o perfil de aliança não tiver a lista de membros carregada,
                // não abrir o formulário como se a captura tivesse sido concluída.
                if (!nomes.length) {
                    alert('Não encontrei a lista de membros desta aliança. Abra a aba/lista de membros, aguarde carregar e tente novamente.');
                    return;
                }
            }

            adicionarNomes(nomes, perfil.tipo === 'alianca'
                ? { tipo: 'alianca', nome: perfil.nome }
                : GRUPO_PERFIL);
            // Encaminha a intenção para o suporte: selecionar este jogador ou
            // os cinco primeiros membros disponíveis da aliança capturada.
            GM_setValue(CHAVE_ACAO_PENDENTE, {
                tipo: perfil.tipo,
                nomes: perfil.tipo === 'jogador' ? [perfil.nome] : nomes,
                grupo: perfil.tipo === 'alianca'
                    ? { tipo: 'alianca', nome: perfil.nome }
                    : (identificarAliancaDoPerfil()
                        ? { tipo: 'alianca', nome: identificarAliancaDoPerfil() }
                        : GRUPO_PERFIL),
                criadoEm: Date.now()
            });
            GM_setValue(CHAVE_SELECAO, []);
            // A lista compartilhada e a ação ficam gravadas antes de abrir o suporte.
            window.open('https://support.innogames.com/playerticket/new', '_blank', 'noopener');
        }

        function atualizarBotaoPerfil() {
            const perfilJogador = encontrarPerfilJogador();
            const perfilAlianca = perfilJogador ? null : encontrarPerfilAlianca();
            const perfil = perfilJogador || perfilAlianca;
            const botaoExistente = document.getElementById(ID_BOTAO);

            if (!perfil) {
                if (botaoExistente) botaoExistente.remove();
                return;
            }

            if (botaoExistente && botaoExistente.dataset.tipo === perfil.tipo &&
                botaoExistente.dataset.nome === perfil.nome) return;
            if (botaoExistente) botaoExistente.remove();

            const botao = document.createElement('button');
            botao.id = ID_BOTAO;
            botao.type = 'button';
            botao.dataset.tipo = perfil.tipo;
            botao.dataset.nome = perfil.nome;
            botao.textContent = perfil.tipo === 'jogador' ? 'Denunciar' : 'Denunciar aliança';
            botao.title = perfil.tipo === 'jogador'
                ? 'Abrir o assistente de denúncia e capturar este jogador'
                : 'Abrir o assistente de denúncia e capturar os membros desta aliança';
            // Reaproveita a estrutura visual nativa dos botões do Grepolis.
            botao.className = 'button_new center gp-denunciar-button' +
                (perfil.tipo === 'jogador' ? ' gp-denunciar-jogador-button' : '');
            botao.style.cssText = perfil.tipo === 'jogador'
                ? 'display:block;clear:both;width:max-content;margin:3px 0 4px;cursor:pointer;transform:scale(0.8);transform-origin:left top;'
                : 'display:inline-block;margin:4px 0;cursor:pointer;';
            botao.innerHTML = '<div class="left"></div><div class="right"></div>' +
                '<div class="caption js-caption"><span>' + botao.textContent +
                '</span><div class="effect js-effect"></div></div>';

            botao.addEventListener('click', () => {
                // Revalidar o perfil no clique para não capturar um nome antigo
                // se a janela tiver mudado entre a renderização e a ação.
                const atual = perfil.tipo === 'jogador' ? encontrarPerfilJogador() : encontrarPerfilAlianca();
                if (!atual) {
                    alert('O perfil não está mais aberto. Abra-o novamente e tente outra vez.');
                    atualizarBotaoPerfil();
                    return;
                }
                abrirSuporteComCaptura(atual);
            });

            // Na aliança, coloca o botão na área nativa de ações (#ally_buttons),
            // ao lado de "Candidatar-se". Para jogadores, usa um botão curto
            // abaixo da linha do cabeçalho e dos ícones do Dio/GRCRT.
            if (perfil.tipo === 'alianca') {
                const areaBotoes = document.querySelector('#ally_buttons .center_box');
                if (areaBotoes) {
                    // Mantém os dois botões na mesma linha e coloca Denunciar
                    // aliança à direita, deslocando Candidatar-se para a esquerda.
                    areaBotoes.style.display = 'flex';
                    areaBotoes.style.flexDirection = 'row';
                    areaBotoes.style.justifyContent = 'center';
                    areaBotoes.style.alignItems = 'center';
                    areaBotoes.style.flexWrap = 'nowrap';
                    areaBotoes.style.gap = '6px';

                    const candidatar = areaBotoes.querySelector('.btn_apply_to_alliance');
                    if (candidatar) {
                        // Neutraliza o posicionamento/alinhamento herdado do botão nativo,
                        // para que ele ocupe o espaço à esquerda sem empurrar o novo botão.
                        candidatar.style.position = 'relative';
                        candidatar.style.left = 'auto';
                        candidatar.style.right = 'auto';
                        candidatar.style.float = 'none';
                        candidatar.style.flex = '0 0 auto';
                        candidatar.style.margin = '4px 0';
                        botao.style.position = 'relative';
                        botao.style.left = 'auto';
                        botao.style.right = 'auto';
                        botao.style.float = 'none';
                        botao.style.flex = '0 0 auto';
                        areaBotoes.insertBefore(candidatar, areaBotoes.firstChild);
                        areaBotoes.insertBefore(botao, candidatar.nextSibling);
                    } else {
                        areaBotoes.appendChild(botao);
                    }
                } else {
                    perfil.raiz.appendChild(botao);
                }
            } else {
                // Para jogador, coloca o botão compacto na faixa azul do título
                // do perfil, ao lado de "Perfil", sem ocupar espaço no conteúdo.
                const janelaPerfil = perfil.raiz.closest('.ui-dialog, .gpwindow') || perfil.raiz.parentElement;
                // O diálogo pode conter outras faixas azuis (por exemplo, a seção de cidades).
                // Prioriza explicitamente o cabeçalho do bloco #profile_info mostrado no perfil.
                const cabecalhoPerfil =
                    janelaPerfil?.querySelector('#profile_info > .game_border > .game_header.bold') ||
                    janelaPerfil?.querySelector('#profile_info .game_header.bold') ||
                    perfil.raiz.closest('#profile_info')?.querySelector('.game_header.bold') ||
                    perfil.raiz.querySelector('.game_header.bold');
                if (cabecalhoPerfil) {
                    cabecalhoPerfil.style.display = 'flex';
                    cabecalhoPerfil.style.alignItems = 'center';
                    cabecalhoPerfil.style.justifyContent = 'space-between';
                    cabecalhoPerfil.style.gap = '8px';
                    botao.style.cssText = 'display:inline-block;flex:0 0 auto;white-space:nowrap;margin:0 4px 0 auto;cursor:pointer;transform:scale(0.968);transform-origin:right center;';
                    cabecalhoPerfil.appendChild(botao);
                } else {
                    // Fallback caso uma variante da interface não tenha o cabeçalho esperado.
                    const grepolisScore = perfil.raiz.querySelector('#player_points .grepolis_score_container');
                    if (grepolisScore) {
                        grepolisScore.insertAdjacentElement('afterend', botao);
                    } else {
                        perfil.raiz.appendChild(botao);
                    }
                }
            }
        }

        const observador = new MutationObserver(() => atualizarBotaoPerfil());
        observador.observe(document.body, { childList: true, subtree: true, attributes: true, attributeFilter: ['class', 'style'] });
        atualizarBotaoPerfil();
    }


    // -------------------------------------------------------------------------

    // -------------------------------------------------------------------------
    // O GrepoData é apenas uma nova fonte de captura. Os nomes entram nas
    // mesmas chaves e no mesmo fluxo já usado pelo sistema Koyo.
    function iniciarIntegracaoGrepoData() {
        const ID_BOTAO_PLAYER = 'gp-grepodata-denunciar-player-koyos';
        const ID_BOTAO_ALLIANCE = 'gp-grepodata-denunciar-alliance-koyos';

        const FLAG_SVG = `
            <svg
                width="14"
                height="14"
                viewBox="0 0 24 24"
                fill="none"
                stroke="currentColor"
                stroke-width="1.5"
                stroke-linecap="round"
                stroke-linejoin="round"
                aria-hidden="true"
                style="width:14px;height:14px;flex:none"
            >
                <path d="M5.25 21V4.5"></path>
                <path d="M5.25 4.5c2.4-1.6 4.8-1.6 7.2 0s4.8 1.6 7.2 0v8.25c-2.4 1.6-4.8 1.6-7.2 0s-4.8-1.6-7.2 0"></path>
            </svg>
        `;

        function criarBotao(texto, id, title) {
            const botao = document.createElement('button');
            botao.id = id;
            botao.type = 'button';
            botao.title = title;
            botao.className =
                'inline-flex items-center gap-1.5 whitespace-nowrap rounded-md border-0 bg-white px-2 py-1 text-xs font-semibold tracking-normal text-slate-900 shadow-sm ring-1 ring-inset ring-slate-300 hover:bg-slate-50 focus-visible:outline focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-brand-700';
            botao.innerHTML = `${FLAG_SVG}<span>${texto}</span>`;
            return botao;
        }

        function aguardarElemento(seletor, limite = 5000) {
            return new Promise(resolve => {
                const inicio = Date.now();

                const verificar = () => {
                    const elemento = document.querySelector(seletor);
                    if (elemento) {
                        resolve(elemento);
                        return;
                    }

                    if (Date.now() - inicio >= limite) {
                        resolve(null);
                        return;
                    }

                    setTimeout(verificar, 100);
                };

                verificar();
            });
        }

        function fecharMailList() {
            const dialog = document.querySelector('maillist-dialog');
            if (!dialog) return;

            const fechar = [...dialog.querySelectorAll('button')]
                .find(botao => /^(close|fechar)$/i.test(botao.textContent.trim()));

            if (fechar) fechar.click();
        }

        async function capturarMailList() {
            const botaoMailList = [...document.querySelectorAll(
                'button[title="Get a list of player names"]'
            )].find(botao => botao.getClientRects().length);

            if (!botaoMailList) {
                throw new Error('Botão "Mail list" não encontrado.');
            }

            let pre = document.querySelector('#bb_code');

            if (!pre) {
                botaoMailList.click();
                pre = await aguardarElemento('#bb_code');
            }

            if (!pre) {
                throw new Error('A lista de nomes do Mail list não foi carregada.');
            }

            const nomes = [...new Set(
                (pre.textContent || '')
                    .split(';')
                    .map(nome => normalizarTexto(nome))
                    .filter(Boolean)
            )];

            if (!nomes.length) {
                throw new Error('O Mail list abriu, mas não contém nomes de jogadores.');
            }

            return nomes;
        }

        function salvarCapturaEPrepararSuporte(nomes, nomeAlianca) {
            adicionarNomes(nomes, {
                tipo: 'alianca',
                nome: nomeAlianca
            });

            GM_setValue(CHAVE_ACAO_PENDENTE, {
                tipo: 'alianca',
                nomes,
                grupo: { tipo: 'alianca', nome: nomeAlianca },
                criadoEm: Date.now()
            });

            GM_setValue(CHAVE_SELECAO, []);

            window.open(
                'https://support.innogames.com/playerticket/new',
                '_blank',
                'noopener'
            );
        }

        async function denunciarAlianca() {
            const botao = document.getElementById(ID_BOTAO_ALLIANCE);
            if (!botao || botao.disabled) return;

            botao.disabled = true;
            const textoOriginal = botao.innerHTML;

        function identificarAliancaDoPerfil() {
            // No perfil do GrepoData, a aliança costuma aparecer como um link
            // para a própria página da aliança. Procuramos links reais de
            // aliança e usamos o texto visível como nome da pasta.
            const links = [...document.querySelectorAll('a[href]')]
                .filter(link => {
                    const href = link.getAttribute('href') || '';
                    let caminho = href;

                    try {
                        caminho = new URL(href, location.origin).pathname;
                    } catch (_) {
                        // Mantém o href original para a verificação abaixo.
                    }

                    return /\/alliance(?:\/|$)/i.test(caminho) &&
                        link.getClientRects().length > 0;
                });

            for (const link of links) {
                const nome = normalizarTexto(link.textContent);
                if (!nome) continue;
                if (/^(alliance|aliança|alliances|alianças)$/i.test(nome)) continue;
                return nome;
            }

            return '';
        }

    function renderizarBuscaAliancas() {
        if (!listaAliancas) return;
        const consulta = String(campoBuscaAliancas?.value || '').trim().toLocaleLowerCase();
        listaAliancas.style.display = consulta ? 'block' : 'none';
        if (!consulta) {
            listaAliancas.replaceChildren();
            return;
        }
        const normalizar = texto => String(texto || '').normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLocaleLowerCase();
        const grupos = (gruposAliançasAtuais || []).filter(grupo =>
            !consulta || normalizar(grupo.nome).includes(normalizar(consulta))
        );
        listaAliancas.replaceChildren();
        if (!grupos.length) {
            listaAliancas.textContent = 'Nenhuma aliança encontrada.';
            return;
        }
        for (const grupo of grupos) {
            const label = document.createElement('label');
            label.style.cssText = 'display:flex;align-items:center;gap:7px;padding:4px 0;cursor:pointer';
            const checkbox = document.createElement('input');
            checkbox.type = 'checkbox';
            checkbox.setAttribute('aria-label', `Selecionar até cinco jogadores de ${grupo.nome}`);
            const nome = document.createElement('span');
            nome.textContent = `${grupo.nome} (${grupo.registros.filter(r => r.tipo === 'jogador' && !nomeEmEspera(r.nome)).length} disponíveis)`;
            checkbox.addEventListener('change', () => {
                if (!checkbox.checked) return;
                const selecionados = obterSelecaoParaSuporte().nomes;
                const existentes = new Set(selecionados.map(chaveJogador));
                const esperas = lerJogadoresEmEspera();
                const candidatos = grupo.registros
                    .filter(r => r.tipo === 'jogador' && !nomeEmEspera(r.nome, esperas) && !existentes.has(chaveJogador(r.nome)))
                    .sort((a, b) => {
                        // Primeiro prioriza contas ainda pendentes; depois, menor
                        // quantidade de reportes registrados; por fim, ordem alfabética.
                        const estadoA = a.estado === 'reportado' ? 1 : 0;
                        const estadoB = b.estado === 'reportado' ? 1 : 0;
                        return estadoA - estadoB ||
                            (Number(a.reportes) || 0) - (Number(b.reportes) || 0) ||
                            a.nome.localeCompare(b.nome, 'pt-BR', { sensitivity: 'base' });
                    });
                let adicionados = 0;
                for (const registro of candidatos) {
                    if (selecionados.length >= 5) break;
                    selecionados.push(registro.nome);
                    existentes.add(chaveJogador(registro.nome));
                    adicionados++;
                }
                checkbox.checked = false;
                GM_setValue(CHAVE_SELECAO_SUPORTE, selecionados);
                atualizarResumoJogadoresSelecionados();
                definirStatus(adicionados
                    ? `${adicionados} jogador(es) selecionado(s) da aliança "${grupo.nome}", priorizando os menos reportados.`
                    : 'Não há jogadores disponíveis dessa aliança para completar a seleção.');
            });
            label.append(checkbox, nome);
            listaAliancas.appendChild(label);
        }
    }

    function lerJogadoresEmEspera() {
        const agora = Date.now();
        let dados = GM_getValue(CHAVE_ESPERA_JOGADORES, {});
        if (!dados || typeof dados !== 'object' || Array.isArray(dados)) dados = {};
        const ativos = {};
        for (const [chave, fim] of Object.entries(dados)) {
            if (Number(fim) > agora) ativos[chave] = Number(fim);
        }
        if (Object.keys(ativos).length !== Object.keys(dados).length) {
            GM_setValue(CHAVE_ESPERA_JOGADORES, ativos);
        }
        return ativos;
    }

    function nomeEmEspera(nome, esperas = lerJogadoresEmEspera()) {
        return Number(esperas[String(nome).toLocaleLowerCase()]) > Date.now();
    }

    // Qualquer jogador ainda em cooldown deve aparecer na subpasta "Já reportados",
    // inclusive os que entraram em espera antes da introdução das pastas.
    function sincronizarReportadosComCooldown(registros = lerJogadoresGuardados()) {
        const esperas = lerJogadoresEmEspera();
        let alterados = false;

        for (const registro of registros) {
            if (nomeEmEspera(registro.nome, esperas) && registro.estado !== 'reportado') {
                registro.estado = 'reportado';
                alterados = true;
            }
        }

        if (alterados) salvarJogadoresGuardados(registros);
        return registros;
    }

    function formatarEspera(fim) {
        const restante = Math.max(0, Number(fim) - Date.now());
        const horas = Math.floor(restante / 3600000);
        const minutos = Math.floor((restante % 3600000) / 60000);
        return `${horas}h ${String(minutos).padStart(2, '0')}min`;
    }

    function obterSelecaoParaSuporte() {
        const nomesBrutos = GM_getValue(CHAVE_SELECAO_SUPORTE, []);
        const nomes = Array.isArray(nomesBrutos)
            ? nomesBrutos.map(nome => String(nome || '').replace(/\s+/g, ' ').trim()).filter(Boolean)
            : [];
        const unicos = [...new Map(nomes.map(nome => [nome.toLocaleLowerCase(), nome])).values()];
        return { nomes: unicos, limite: 5, valido: unicos.length >= 1 && unicos.length <= 5 };
    }

                .toLocaleLowerCase()

    function atualizarMensagemMulticontas() {
        const selecao = obterSelecaoParaSuporte();
        if (!selecao.valido) {
            atualizarResumoJogadoresSelecionados();
            definirStatus(
                `Selecione de 1 a 5 jogadores na lista do suporte para preparar este lote.`
            );
            return;
        }

        atualizarResumoJogadoresSelecionados();
        const tipo = TIPOS_RELATORIO[tipoAtivo] || TIPOS_RELATORIO.multicontas;
        const partes = [tipo.mensagemBase.trimEnd()];

        for (const [id, categoria] of Object.entries(SUBCATEGORIAS)) {
            if (tipo.categorias.includes(id) && categoriasSelecionadas.has(id)) {
                partes.push(categoria.mensagem);
            }
        }

        tituloInput.value = tipo.titulo;
        mensagemInput.value = partes.join('\n\n');

        definirStatus(
            `${selecao.nomes.length} jogador(es) incluído(s). ` +
            (categoriasSelecionadas.size
                ? `${categoriasSelecionadas.size} tipo(s) de denúncia selecionado(s).`
                : 'Modelo de multicontas preparado.'),
            true
        );
    }

    function abrirTipoRelatorio(tipo) {
        tipoAtivo = tipo;
        const configuracao = TIPOS_RELATORIO[tipo];
        categoriasSelecionadas.clear();
        for (const item of listaCategorias.children) {
            const visivel = configuracao.categorias.includes(item.dataset.categoria);
            item.hidden = !visivel;
            const botaoCategoria = item.querySelector('button[aria-pressed]');
            if (botaoCategoria) {
                botaoCategoria.setAttribute('aria-pressed', 'false');
                botaoCategoria.style.background = '#fff';
                botaoCategoria.style.borderColor = '#bbb';
                botaoCategoria.style.fontWeight = 'normal';
                botaoCategoria.setAttribute('aria-expanded', 'false');
                const dica = item.querySelector('div');
                if (dica && (SUBCATEGORIAS[item.dataset.categoria].dica || SUBCATEGORIAS[item.dataset.categoria].dicaMulticontas)) {
                    dica.hidden = true;
                    dica.textContent = tipo === 'multicontas'
                        ? (SUBCATEGORIAS[item.dataset.categoria].dicaMulticontas || SUBCATEGORIAS[item.dataset.categoria].dica || '')
                        : (SUBCATEGORIAS[item.dataset.categoria].dica || '');
                }
            }
        }
        for (const botao of botoesTipo) {
            const ativo = botao.dataset.tipo === tipo;
            botao.setAttribute('aria-expanded', String(ativo));
            botao.style.background = ativo ? '#e5f5e9' : '';
            botao.style.borderColor = ativo ? '#18794e' : '';
        }
        submenu.hidden = false;
        atualizarMensagemMulticontas();
    }

    for (const botao of botoesTipo) {
        botao.addEventListener('click', () => {
            const tipo = botao.dataset.tipo;
            if (tipoAtivo === tipo && !submenu.hidden) {
                submenu.hidden = true;
                tipoAtivo = null;
                botao.setAttribute('aria-expanded', 'false');
                return;
            }
            abrirTipoRelatorio(tipo);
        });
    }

    // Alerta sonoro original.
    function tocarAlerta() {
        try {
            const AudioContextClass =
                window.AudioContext || window.webkitAudioContext;

            if (!AudioContextClass) {
                timerStatus.textContent =
                    'Tempo concluído! Seu navegador não permite o alerta sonoro.';
                return;
            }

            const audio = new AudioContextClass();

            audio.resume().then(() => {
                [0, 0.35, 0.7].forEach((atraso) => {
                    const oscilador = audio.createOscillator();
                    const volume = audio.createGain();

                    oscilador.type = 'sine';
                    oscilador.frequency.value = 880;
                    volume.gain.value = 0.18;

                    oscilador.connect(volume);
                    volume.connect(audio.destination);

                    oscilador.start(audio.currentTime + atraso);
                    oscilador.stop(audio.currentTime + atraso + 0.18);
                });

                setTimeout(() => audio.close(), 1500);
            }).catch(() => {
                timerStatus.textContent =
                    'Tempo concluído! Clique em Testar som para habilitar o áudio.';
            });
        } catch (erro) {
            console.warn(
                '[Grepolis] Não foi possível tocar o alerta:',
                erro
            );
        }
    }

    // Temporizador original.
    function atualizarTimer() {
        const fim = Number(sessionStorage.getItem(CHAVE_TIMER));

        if (!fim) {
            timerStatus.textContent = 'Temporizador inativo.';
            return;
        }

        const restante = Math.max(0, fim - Date.now());

        if (restante <= 0) {
            clearInterval(timerInterval);
            timerInterval = null;

            sessionStorage.removeItem(CHAVE_TIMER);

            timerStatus.textContent =
                'Tempo concluído! Você já pode voltar ao formulário.';

            timerStatus.style.background = '#e5f5e9';
            timerStatus.style.color = '#18794e';

            if (!avisoDetectado) {
                avisoDetectado = true;
                tocarAlerta();
                document.title = '⏰ Espera concluída — Grepolis';
            }

            return;
        }

        const minutos = Math.floor(restante / 60000);
        const segundos = Math.floor((restante % 60000) / 1000);

        timerStatus.textContent =
            `Aguarde ${String(minutos).padStart(2, '0')}:` +
            `${String(segundos).padStart(2, '0')}`;

        timerStatus.style.background = '#fff3cd';
        timerStatus.style.color = '#8a5800';
    }

    function iniciarTimer() {
        if (!sessionStorage.getItem(CHAVE_TIMER)) {
            sessionStorage.setItem(
                CHAVE_TIMER,
                String(Date.now() + DURACAO_ESPERA)
            );
        }

        avisoDetectado = false;

        if (timerInterval) clearInterval(timerInterval);

        atualizarTimer();
        timerInterval = setInterval(atualizarTimer, 1000);
    }

    // Adiciona os jogadores sequencialmente, aguardando o formulário processar cada inclusão.
    // A lista oficial #players_string é usada para confirmar cada resultado.
    const PAUSA_ENTRE_JOGADORES = 3000;
    const TEMPO_MAXIMO_CONFIRMACAO = 5000;

    function aguardar(ms) {
        return new Promise(resolve => setTimeout(resolve, ms));
    }

    function normalizarNome(valor) {
        return String(valor || '').trim().toLocaleLowerCase();
    }

    async function aguardarNomeNaLista(nome, lista, nomesAntes) {
        const inicio = Date.now();
        const nomeNormalizado = normalizarNome(nome);

        while (Date.now() - inicio < TEMPO_MAXIMO_CONFIRMACAO) {
            const textoAtual = lista.textContent || '';
            const nomesAtuais = textoAtual.split(/[\n,;|]+/).map(normalizarNome).filter(Boolean);

            // Confirma uma mudança na lista e a presença do nome solicitado.
            if (nomesAtuais.some(item => item === nomeNormalizado) ||
                (normalizarNome(textoAtual).includes(nomeNormalizado) &&
                 normalizarNome(textoAtual) !== normalizarNome(nomesAntes))) {
                return true;
            }

            await aguardar(250);
        }

        return false;
    }

    async function adicionarJogadoresAoFormulario(formulario) {
        const selecao = obterSelecaoParaSuporte();
        if (!selecao.valido) {
            definirStatus(`Seleção inválida: são necessários exatamente ${selecao.limite} jogador(es).`);
            return false;
        }

        const campoNome = formulario.querySelector('#player_name');
        const botaoAdicionar = campoNome?.closest('.input-group')?.querySelector('button.add')
            || formulario.querySelector('button.add');

        if (!campoNome || !botaoAdicionar) {
            definirStatus('Título e mensagem foram localizados, mas não encontrei o campo #player_name e o botão Adicionar do formulário.');
            return false;
        }

        const listaJogadores = formulario.querySelector('#players_string') || document.querySelector('#players_string');
        if (!listaJogadores) {
            definirStatus('Não encontrei #players_string para confirmar as inclusões. Nenhum jogador foi adicionado automaticamente.');
            return false;
        }

        definirStatus(`Adicionando nomes... (0/${selecao.nomes.length})`, false);

        for (let i = 0; i < selecao.nomes.length; i++) {
            const nome = selecao.nomes[i];
            const nomesAntes = listaJogadores.textContent || '';

            // Não clica novamente se o nome já estiver registrado no formulário.
            if (normalizarNome(nomesAntes).includes(normalizarNome(nome))) {
                definirStatus(`Adicionando nomes... (${i + 1}/${selecao.nomes.length}) — ${nome} já estava na lista.`, true);
                continue;
            }

            definirStatus(`Adicionando nomes... (${i + 1}/${selecao.nomes.length}) — ${nome}`, false);
            campoNome.value = nome;
            campoNome.dispatchEvent(new Event('input', { bubbles: true }));
            campoNome.dispatchEvent(new Event('change', { bubbles: true }));
            botaoAdicionar.click();

            const confirmado = await aguardarNomeNaLista(nome, listaJogadores, nomesAntes);
            if (!confirmado) {
                definirStatus(`A inclusão de "${nome}" não foi confirmada em até ${TEMPO_MAXIMO_CONFIRMACAO / 1000} segundos. O processo foi interrompido para evitar continuar com falhas.`);
                return false;
            }

            incrementarContagemReportes(nome);
            definirStatus(`Adicionando nomes... (${i + 1}/${selecao.nomes.length}) — ${nome} adicionado.`, true);

            // Dá ao site uma margem adicional antes da próxima interação.
            if (i < selecao.nomes.length - 1) {
                await aguardar(PAUSA_ENTRE_JOGADORES);
            }
        }

        // Coloca em espera por 6 horas apenas após confirmar o lote inteiro.
        const esperas = lerJogadoresEmEspera();
        const fimEsperaJogadores = Date.now() + DURACAO_ESPERA_JOGADOR;
        for (const nome of selecao.nomes) {
            esperas[nome.toLocaleLowerCase()] = fimEsperaJogadores;
        }
        GM_setValue(CHAVE_ESPERA_JOGADORES, esperas);
        marcarJogadoresComoReportados(selecao.nomes);
        GM_setValue(CHAVE_SELECAO_SUPORTE, []);
        atualizarResumoJogadoresSelecionados();
        return true;
    }

    // Preenchimento do formulário oficial.
    async function preencherCamposSemEventos() {
        const formulario = document.querySelector('#submit_form');
        const titulo = formulario?.querySelector('input[name="subject"]');
        const mensagem = formulario?.querySelector('textarea#message');

        if (!formulario || !titulo || !mensagem) {
            definirStatus('Não foi possível encontrar título ou mensagem.');
            return false;
        }

        titulo.value = tituloInput.value;
        mensagem.value = mensagemInput.value;
        const quantidadeSelecionada = obterSelecaoParaSuporte().nomes.length;

        const jogadoresAdicionados = await adicionarJogadoresAoFormulario(formulario);
        const sucesso =
            titulo.value === tituloInput.value &&
            mensagem.value === mensagemInput.value &&
            jogadoresAdicionados;

        definirStatus(
            sucesso
                ? `Título e mensagem preenchidos; lote de até ${quantidadeSelecionada} jogador(es) processado. O temporizador de 5 minutos foi iniciado.`
                : 'Não foi possível confirmar o preenchimento de todos os campos ou a adição dos jogadores.',
            sucesso
        );

        if (sucesso) iniciarTimer();
        return sucesso;
    }

    function obterCategoriaRegras() {
        const categoria = document.querySelector(
            '#submit_form select[name="typeID"]'
        );

        if (!categoria) return null;

        const opcao = [...categoria.options].find(option =>
            /quebra de regras/i.test(option.textContent.trim())
        );

        return opcao ? { campo: categoria, opcao } : null;
    }

    async function preencher() {
        const resultado = obterCategoriaRegras();

        if (!resultado) {
            definirStatus(
                'Campo de categoria ou opção "Quebra de Regras" não encontrado.'
            );
            return;
        }

        const { campo, opcao } = resultado;

        if (campo.value !== opcao.value) {
            definirStatus(
                'A categoria ainda não está selecionada. Aguarde a atualização da página e tente novamente.'
            );

            campo.value = opcao.value;

            campo.dispatchEvent(
                new Event('change', { bubbles: true })
            );

            return;
        }

        await preencherCamposSemEventos();
    }

    botaoPreencher.addEventListener('click', async event => {
        event.preventDefault();
        event.stopImmediatePropagation();
        if (botaoPreencher.disabled) return;
        const fimEspera = Number(sessionStorage.getItem(CHAVE_TIMER) || 0);
        if (fimEspera > Date.now()) {
            definirStatus('O assistente está em espera após o lote anterior. Aguarde o fim dos 5 minutos para adicionar outro lote ou preparar outra denúncia.');
            atualizarTimer();
            return;
        }
        botaoPreencher.disabled = true;
        const textoOriginalBotao = botaoPreencher.textContent;
        botaoPreencher.textContent = 'Adicionando nomes...';
        definirStatus('Adicionando nomes... preparando o formulário.', false);
        try {
            await preencher();
        } catch (erro) {
            console.error('[Koyo] Falha ao preencher jogadores:', erro);
            definirStatus('O preenchimento foi interrompido por um erro. Confira a lista de jogadores antes de tentar novamente.');
        } finally {
            botaoPreencher.disabled = false;
            botaoPreencher.textContent = textoOriginalBotao;
        }
    }, true);

    botaoSom.addEventListener('click', tocarAlerta);

    // O temporizador do assistente começa após a inclusão bem-sucedida do lote,
    // independentemente do botão oficial de envio do site.

    // Seleciona "Quebra de Regras" durante a inicialização.
    // A chave de sessão evita repetir a tentativa indefinidamente
    // se a página atualizar sem preservar a seleção.
    function inicializarCategoria() {
        const resultado = obterCategoriaRegras();

        if (!resultado) return;

        const { campo, opcao } = resultado;
        const tentativaAnterior =
            sessionStorage.getItem(CHAVE_CATEGORIA_INICIAL) === '1';

        if (campo.value === opcao.value) {
            sessionStorage.removeItem(CHAVE_CATEGORIA_INICIAL);
            return;
        }

        if (tentativaAnterior) {
            sessionStorage.removeItem(CHAVE_CATEGORIA_INICIAL);
            return;
        }

        sessionStorage.setItem(CHAVE_CATEGORIA_INICIAL, '1');

        campo.value = opcao.value;

        campo.dispatchEvent(
            new Event('change', { bubbles: true })
        );
    }

    sincronizarReportadosComCooldown();
    atualizarResumoJogadoresSelecionados();
    inicializarCategoria();

    // Restaura o temporizador se a página for recarregada.
    if (sessionStorage.getItem(CHAVE_TIMER)) {
        atualizarTimer();

        if (sessionStorage.getItem(CHAVE_TIMER)) {
            timerInterval = setInterval(atualizarTimer, 1000);
        }
    }
})();
